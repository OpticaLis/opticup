On Error GoTo out

    
    Dim sw As Integer
    
    
    
    If Date < qdates Then
     sw = -99
    Else
      sw = DateDiff("d", qdates, Date)
    End If
     
    
     

    q30 = sw
    
   
out:
     
End Function
