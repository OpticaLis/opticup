Function ChangeProperty(strPropName As String, _
    varPropType As Variant, _
    varPropValue As Variant) As Integer

    Dim dbs As Object, prp As Variant
    Const conPropNotFoundError = 3270

    Set dbs = CurrentDb
    On Error GoTo Change_Err
    dbs.Properties(strPropName) = varPropValue
    ChangeProperty = True

Change_Bye:
    Exit Function

Change_Err:
    If err = conPropNotFoundError Then ' Property not found.
        Set prp = dbs.CreateProperty(strPropName, _
            varPropType, varPropValue)
        dbs.Properties.Append prp
        Resume Next
    Else
        ' Unknown error.
        ChangeProperty = False
        Resume Change_Bye
    End If
End Function

Function xg_ReplaceAllWith(sMainString As Variant, _
sSubString As String, sReplaceString As String) As Variant
'* Recursive function to replace all occurences of sSubString
'* with sReplaceString in sMainString
Dim i As Integer
Dim ipos As Integer
Dim s As Variant
Dim s1 As Variant, s2 As Variant
On Error GoTo out
If Nz(sMainString) = "" Then
 Exit Function
End If
s = sMainString
ipos = InStr(1, sMainString, sSubString)
If ipos = 0 Then
    GoTo Exit_xg_ReplaceAllWith
End If
s1 = Mid(sMainString, 1, ipos - 1)
s2 = Mid(sMainString, ipos + Len(sSubString), Len(sMainString))
s = s1 & sReplaceString & _
    xg_ReplaceAllWith(s2, sSubString, sReplaceString)

Exit_xg_ReplaceAllWith:
    xg_ReplaceAllWith = s
    Exit Function
out:
    xg_ReplaceAllWith = Nz(sMainString)
End Function

Function xg_GetSubString(mainstr As String, n As Integer, delimiter As String) As String
'* Get the "n"-th substring from "mainstr" where strings are delimited by "delimiter"
    Dim i As Integer
    Dim substringcount As Integer
    Dim pos As Integer
    Dim strx As String
    Dim val1 As Integer
    Dim w As String

On Error GoTo Err_xg_GetSubString

w = ""
substringcount = 0
i = 1
pos = InStr(i, mainstr, delimiter)
Do While pos <> 0
    strx = Mid(mainstr, i, pos - i)
    substringcount = substringcount + 1
    If substringcount = n Then
        Exit Do
    End If
    i = pos + 1
    pos = InStr(i, mainstr, delimiter)
Loop

If substringcount = n Then
    xg_GetSubString = strx
Else
    strx = Mid(mainstr, i, Len(mainstr) + 1 - i)
    substringcount = substringcount + 1
    If substringcount = n Then
        xg_GetSubString = strx
    Else
        xg_GetSubString = ""
    End If
End If

Exit Function

Err_xg_GetSubString:
    MsgBox "xg_GetSubString " & err & " " & err.Description
    Resume Next

End Function

Function last_GetSubString(mainstr As String, delimiter As String) As String
'* Get the last substring from "mainstr" where strings are delimited by "delimiter"
    Dim i As Integer
    
    Dim pos As Integer
    Dim strx As String
    
    

On Error GoTo Err_xg_GetSubString
If Nz(mainstr) <> "" Then
 mainstr = mainstr & delimiter
End If
strx = ""
i = 1
pos = InStr(i, mainstr, delimiter)
Do While pos <> 0
    strx = Mid(mainstr, i, pos - i)
   
    If Nz(strx) = "" Then
        i = pos - 1
        pos = InStr(i, mainstr, delimiter)
        strx = Mid(mainstr, i, pos - i)
        Exit Do
    End If
    i = pos + 1
    pos = InStr(i, mainstr, delimiter)
Loop

 
    last_GetSubString = Nz(strx)


Exit Function

Err_xg_GetSubString:
    MsgBox "last_GetSubString " & err & " " & err.Description
    Resume Next

End Function

Public Function getFileName() As String
 On Error GoTo out

  Dim f    As Object
  Dim buff As Variant
  Dim folder As Variant
  Dim X%, Y As Integer
 
      
   buff = ""
      
   Set f = Application.FileDialog(3)
   f.AllowMultiSelect = True
   Set dbs = CurrentDb()
   Set rst = dbs.OpenRecordset("tb_lastfolder")
   rst.MoveFirst
   
    If Nz(rst![LastFolder]) <> "" Then
      f.InitialFileName = Nz(rst![LastFolder])
    End If
   f.Show
   On Error Resume Next
   buff = Trim$(Nz(f.SelectedItems(1)))
   On Error GoTo out


  
    If Nz(buff) <> "" Then
      Y = 999
      X = 1
      Do Until Y = 0
         Y = InStr(X, buff, "\")
         If Y <> 0 Then
          X = Y + 1
         End If
      Loop
      folder = Mid(buff, 1, X - 1)
      rst.Edit
      rst![LastFolder] = folder
      rst.Update
      rst.Close
    
       getFileName = buff
   
  End If

out:
End Function
