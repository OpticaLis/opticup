Function calc_hours(intm) As String

Dim tots As Double

  tots = ((intm / 60))
  ii = Int([tots])
  aa = intm - ([ii] * 60)
Else
   ii = 0
   aa = intm
End If

If aa > 0 Then
 If aa < 10 Then
    calc_hours = Trim$(str$([ii])) + ":" + "0" + Trim$(str$([aa]))
